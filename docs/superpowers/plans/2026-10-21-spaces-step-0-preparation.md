# Spaces, step 0: preparation — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Free the word "space" by renaming the physical sites to places everywhere (ADR 0007), and let members edit their own profile on the website.

**Architecture:** Part A renames the `spaces` collection to `places` and the `space` relationships to `place`, in code, seed, import scripts, tests and the database. The migration is generated by Payload (for its schema snapshot) and its SQL is replaced by hand-written `RENAME` statements, so no row is lost. Part B adds a signed-in `/{locale}/profile` page: a server module reads and writes the member's own person through Payload's access rules, and the bio moves between Lexical (stored) and Markdown (edited) with Payload's converters.

**Tech Stack:** Payload 3.91 (Postgres adapter, `push: false`), Next.js 16 App Router, next-intl, Arc components, zod 4, Vitest.

**Spec:** `docs/superpowers/specs/2026-10-21-spaces-design.md` (section 14, step 0).

## Ground rules for this repo

- Work on `main`; no feature branch.
- Never run `pnpm db:migrate:create` or `pnpm db:migrate` yourself. Ask the user to run them with `! <command>` and wait.
- Integration tests run against the real `DATABASE_URL` and need the migrated schema: `pnpm exec vitest run --config ./vitest.config.mts <file>`.
- Known failures that are not yours: `publishing.int.spec.ts › duplicates an approved article as a fresh draft` fails on `main` already. `applications` and `import-media` int files fail only when the whole int folder runs at once; they pass alone.

- Format with `pnpm exec prettier --write <files>`, lint with `pnpm exec eslint <files>`, typecheck with `pnpm exec tsc --noEmit -p .`.

import {
  convertLexicalToMarkdown,
  convertMarkdownToLexical,
  editorConfigFactory,
} from '@payloadcms/richtext-lexical'
import type { SerializedEditorState } from '@payloadcms/richtext-lexical/lexical'
import type { Payload } from 'payload'
import { z } from 'zod'

import { personOf } from '@/access'
import { SOCIAL_PLATFORMS } from '@/fields/social'
import type { User } from '@/payload-types'

type Locale = 'en' | 'ar'
type SocialLink = { platform: (typeof SOCIAL_PLATFORMS)[number]; url: string }

/** What the profile page shows and edits: the member's own person. */
export type ProfileView = {
  personId: number
  /** Whether the person is on the public site. Only an admin changes it. */
  visible: boolean
  name: Record<Locale, string>
  /** The bio as Markdown, per language. */
  bio: Record<Locale, string>
  website: string
  socialLinks: SocialLink[]
  photo: { url: string; alt: string } | null
}

export type ProfileInput = {
  name: Record<Locale, string>
  bio: Record<Locale, string>
  website: string
  socialLinks: SocialLink[]
}

/** A field the page names in its error message, or the account has no person. */
export type ProfileError = 'name' | 'bio' | 'website' | 'social' | 'photo' | 'no-person'

const httpUrl = z.string().trim().url().regex(/^https?:\/\//i)

const schema = z.object({
  name: z.object({ en: z.string().trim().min(1).max(120), ar: z.string().trim().max(120) }),
  bio: z.object({ en: z.string().max(5000), ar: z.string().max(5000) }),
  website: z.union([z.literal(''), httpUrl]),
  socialLinks: z.array(z.object({ platform: z.enum(SOCIAL_PLATFORMS), url: httpUrl })).max(8),
})

const PHOTO_TYPES = ['image/jpeg', 'image/png', 'image/webp']
const PHOTO_MAX_BYTES = 5 * 1024 * 1024

const editorConfig = (payload: Payload) => editorConfigFactory.default({ config: payload.config })

const toMarkdown = async (payload: Payload, data: unknown): Promise<string> =>
  data && typeof data === 'object' && 'root' in data
    ? convertLexicalToMarkdown({
        data: data as SerializedEditorState,
        editorConfig: await editorConfig(payload),
      }).trim()
    : ''

const toLexical = async (payload: Payload, markdown: string) =>
  markdown.trim()
    ? convertMarkdownToLexical({ editorConfig: await editorConfig(payload), markdown })
    : null

const textOf = (value: unknown, locale: Locale): string =>
  value && typeof value === 'object' ? String((value as Record<string, unknown>)[locale] ?? '') : ''

/**
 * The signed-in member's own person, as the profile page edits it, or null
 * when the account has no person. Read through access rules, as the member.
 */
export const readProfile = async (payload: Payload, user: User): Promise<ProfileView | null> => {
  const id = personOf(user)
  if (id === null) return null
  const person = await payload.findByID({
    collection: 'people',
    id,
    locale: 'all',
    depth: 1,
    user,
    overrideAccess: false,
  })
  const bio = (person as { bio?: unknown }).bio
  const photo = typeof person.photo === 'object' && person.photo?.url ? person.photo : null
  return {
    personId: person.id,
    visible: Boolean(person.visible),
    name: { en: textOf(person.name, 'en'), ar: textOf(person.name, 'ar') },
    bio: {
      en: await toMarkdown(payload, (bio as Record<string, unknown> | undefined)?.en),
      ar: await toMarkdown(payload, (bio as Record<string, unknown> | undefined)?.ar),
    },
    website: person.website ?? '',
    socialLinks: (person.social_links ?? []).map(({ platform, url }) => ({ platform, url })),
    photo: photo ? { url: photo.url!, alt: photo.alt } : null,
  }
}

/**
 * Saves the member's own name, bio, links and photo. Everything goes through
 * the collection's access rules as the member: `visible` is admin-only there,
 * so it is never written from here, whatever the input holds.
 */
export const saveProfile = async (
  payload: Payload,
  user: User,
  input: ProfileInput,
  photo?: { data: Buffer; mimetype: string; name: string; size: number },
): Promise<{ ok: true } | { ok: false; error: ProfileError }> => {
  const id = personOf(user)
  if (id === null) return { ok: false, error: 'no-person' }

  const parsed = schema.safeParse(input)
  if (!parsed.success) {
    const field = String(parsed.error.issues[0]?.path[0] ?? 'name')
    return { ok: false, error: (field === 'socialLinks' ? 'social' : field) as ProfileError }
  }
  const { name, bio, website, socialLinks } = parsed.data

  let photoId: number | undefined
  if (photo) {
    if (!PHOTO_TYPES.includes(photo.mimetype) || photo.size > PHOTO_MAX_BYTES) {
      return { ok: false, error: 'photo' }
    }
    const media = await payload.create({
      collection: 'media',
      data: { alt: name.en },
      file: photo,
      user,
      overrideAccess: false,
    })
    photoId = media.id
  }

  const asMember = { collection: 'people', id, user, overrideAccess: false } as const
  await payload.update({
    ...asMember,
    locale: 'en',
    data: {
      name: name.en,
      bio: await toLexical(payload, bio.en),
      website: website || null,
      social_links: socialLinks,
      ...(photoId ? { photo: photoId } : {}),
    },
  })
  await payload.update({
    ...asMember,
    locale: 'ar',
    // An empty Arabic name falls back to the English one on the site.
    data: { ...(name.ar ? { name: name.ar } : {}), bio: await toLexical(payload, bio.ar) },
  })
  return { ok: true }
}

'use server'

import config from '@payload-config'
import { getPayload } from 'payload'

import { saveProfile, type ProfileError, type ProfileInput } from '@/lib/profile'
import { currentReader } from '@/lib/reader'

export type ProfileState = { status: 'idle' | 'saved' } | { status: 'error'; error: ProfileError }

const text = (form: FormData, key: string) => String(form.get(key) ?? '')

/** The profile form's submit: the signed-in member saves their own person. */
export async function saveProfileAction(
  _previous: ProfileState,
  form: FormData,
): Promise<ProfileState> {
  const user = await currentReader()
  if (!user) return { status: 'error', error: 'no-person' }

  let socialLinks: ProfileInput['socialLinks'] = []
  try {
    socialLinks = JSON.parse(text(form, 'socialLinks') || '[]')
  } catch {
    return { status: 'error', error: 'social' }
  }
  const input: ProfileInput = {
    name: { en: text(form, 'name_en'), ar: text(form, 'name_ar') },
    bio: { en: text(form, 'bio_en'), ar: text(form, 'bio_ar') },
    website: text(form, 'website'),
    socialLinks,
  }
  const file = form.get('photo')
  const photo =
    file instanceof File && file.size > 0
      ? {
          data: Buffer.from(await file.arrayBuffer()),
          mimetype: file.type,
          name: file.name,
          size: file.size,
        }
      : undefined

  const payload = await getPayload({ config })
  const result = await saveProfile(payload, user, input, photo)
  return result.ok ? { status: 'saved' } : { status: 'error', error: result.error }
}

```tsx
'use client'

import { useActionState, useState } from 'react'

import type { ProfileState } from '@/app/(frontend)/[locale]/profile/actions'
import { Button } from '@/components/arc/button/button'
import { Input } from '@/components/arc/input/input'
import { RichTextEditor } from '@/components/arc/rich-text-editor/rich-text-editor'
import { SOCIAL_PLATFORMS } from '@/fields/social'
import type { ProfileError, ProfileView } from '@/lib/profile'

export type ProfileMessages = Record<
  | 'public' | 'private' | 'nameEn' | 'nameAr' | 'bioEn' | 'bioAr' | 'website' | 'social'
  | 'platform' | 'url' | 'addLink' | 'removeLink' | 'photo' | 'photoHint' | 'save' | 'saved',
  string
> & { errors: Record<ProfileError, string> }

type Props = {
  profile: ProfileView
  messages: ProfileMessages
  action: (state: ProfileState, form: FormData) => Promise<ProfileState>
}

/**
 * The member's own profile. The bios are edited as Markdown with Arc's rich
 * text editor and sent in hidden fields; the server stores them as Lexical.
 */
export function ProfileForm({ profile, messages: m, action }: Props) {
  const [state, submit, pending] = useActionState(action, { status: 'idle' } as ProfileState)
  const [bioEn, setBioEn] = useState(profile.bio.en)
  const [bioAr, setBioAr] = useState(profile.bio.ar)
  const [links, setLinks] = useState(profile.socialLinks)

  return (
    <form action={submit} className="grid gap-6">
      <p className="text-sm text-muted-foreground">{profile.visible ? m.public : m.private}</p>

      <Input label={m.nameEn} name="name_en" defaultValue={profile.name.en} required dir="ltr" />
      <Input label={m.nameAr} name="name_ar" defaultValue={profile.name.ar} dir="rtl" />

      <fieldset className="grid gap-2">
        <legend className="text-sm font-medium">{m.bioEn}</legend>
        <div dir="ltr">
          <RichTextEditor
            aria-label={m.bioEn}
            defaultMarkdown={profile.bio.en}
            onChange={(value) => setBioEn(value.markdown)}
          />
        </div>
        <input type="hidden" name="bio_en" value={bioEn} />
      </fieldset>
      <fieldset className="grid gap-2">
        <legend className="text-sm font-medium">{m.bioAr}</legend>
        <div dir="rtl">
          <RichTextEditor
            aria-label={m.bioAr}
            defaultMarkdown={profile.bio.ar}
            onChange={(value) => setBioAr(value.markdown)}
          />
        </div>
        <input type="hidden" name="bio_ar" value={bioAr} />
      </fieldset>

      <Input label={m.website} name="website" type="url" defaultValue={profile.website} dir="ltr" />

      <fieldset className="grid gap-3">
        <legend className="text-sm font-medium">{m.social}</legend>
        {links.map((link, i) => (
          <div key={i} className="flex flex-wrap items-end gap-2">
            <label className="grid gap-1 text-sm">
              {m.platform}
              <select
                className="rounded-md border px-2 py-2"
                value={link.platform}
                onChange={(e) =>
                  setLinks(links.map((l, j) => (j === i ? { ...l, platform: e.target.value as typeof l.platform } : l)))
                }
              >
                {SOCIAL_PLATFORMS.map((p) => (
                  <option key={p} value={p}>
                    {p.charAt(0).toUpperCase() + p.slice(1)}
                  </option>
                ))}
              </select>
            </label>
            <div className="min-w-60 flex-1">
              <Input
                label={m.url}
                type="url"
                dir="ltr"
                value={link.url}
                onChange={(e) => setLinks(links.map((l, j) => (j === i ? { ...l, url: e.target.value } : l)))}
              />
            </div>
            <Button type="button" variant="ghost" onClick={() => setLinks(links.filter((_, j) => j !== i))}>
              {m.removeLink}
            </Button>
          </div>
        ))}
        {links.length < 8 && (
          <Button
            type="button"
            variant="secondary"
            onClick={() => setLinks([...links, { platform: 'instagram', url: '' }])}
          >
            {m.addLink}
          </Button>
        )}
        <input type="hidden" name="socialLinks" value={JSON.stringify(links)} />
      </fieldset>

      <div className="grid gap-2">
        {profile.photo && (
          // eslint-disable-next-line @next/next/no-img-element -- a member's own upload, any size
          <img src={profile.photo.url} alt={profile.photo.alt} className="size-24 rounded-full object-cover" />
        )}
        <Input
          label={m.photo}
          description={m.photoHint}
          name="photo"
          type="file"
          accept="image/jpeg,image/png,image/webp"
        />
      </div>

      <div className="flex items-center gap-3">
        <Button type="submit" loading={pending}>
          {m.save}
        </Button>
        <p role="status" className="text-sm">
          {state.status === 'saved' ? m.saved : state.status === 'error' ? m.errors[state.error] : ''}
        </p>
      </div>
    </form>
  )
}
```

**Files:** Create: `src/app/(frontend)/[locale]/profile/page.tsx`

- [ ] **Step 1: Write the page**

```tsx
import config from '@payload-config'
import type { Metadata } from 'next'
import { getTranslations } from 'next-intl/server'
import { redirect } from 'next/navigation'
import { getPayload } from 'payload'

import { ProfileForm, type ProfileMessages } from '@/components/profile/profile-form'
import { readProfile } from '@/lib/profile'
import { currentReader } from '@/lib/reader'

import { saveProfileAction } from './actions'

type Props = { params: Promise<{ locale: string }> }

export async function generateMetadata(): Promise<Metadata> {
  const t = await getTranslations('Profile')
  return { title: t('metaTitle'), robots: { index: false } }
}

/** The signed-in member's own profile (spec: Spaces, step 0). */
export default async function ProfilePage({ params }: Props) {
  const { locale } = await params
  const user = await currentReader()
  if (!user) redirect(`/${locale}/login?next=${encodeURIComponent(`/${locale}/profile`)}`)

  const t = await getTranslations('Profile')
  const profile = await readProfile(await getPayload({ config }), user)

  const keys = [
    'public', 'private', 'nameEn', 'nameAr', 'bioEn', 'bioAr', 'website', 'social',
    'platform', 'url', 'addLink', 'removeLink', 'photo', 'photoHint', 'save', 'saved',
  ] as const
  const errors = ['name', 'name-ar', 'bio', 'website', 'social', 'photo', 'no-person'] as const
  const messages = {
    ...Object.fromEntries(keys.map((k) => [k, t(k)])),
    errors: Object.fromEntries(errors.map((k) => [k, t(`errors.${k}`)])),
  } as ProfileMessages

  return (
    <div className="mx-auto grid max-w-2xl gap-6 px-4 py-12">
      <header className="grid gap-2">
        <h1 className="text-3xl font-medium">{t('title')}</h1>
        <p className="text-muted-foreground">{t('intro')}</p>
      </header>
      {profile ? (
        <ProfileForm profile={profile} messages={messages} action={saveProfileAction} />
      ) : (
        <p>{t('noPerson')}</p>
      )}
    </div>
  )
}
```
