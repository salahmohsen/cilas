# Emails Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** One branded email layout for every email CILAS sends, texts that admins edit in an Email settings global, and newsletter issues that admins build from courses, articles and programmes, preview live, test, and send through Resend Broadcasts.

**Architecture:** React Email components (styles adapted from the MIT "Barebone" demo, as inline style objects, no Tailwind) render to HTML and plain text in `src/emails/`. An `email-settings` global holds the logo, accent, sender, footer and every transactional text with `{placeholders}`; code reads it through one `getEmailSettings()` with defaults. Newsletters gain `picks` (polymorphic relationship + note), a live preview route that returns the real email HTML, and two admin endpoints: send a test with Payload's `sendEmail`, and send the issue with Resend's REST `POST /broadcasts`.

**Tech Stack:** Payload 3.91, Next.js, `react-email` 6.x (`render`, components), Resend REST API, Lexical HTML converter, Vitest.

**Decisions already made (do not reopen):**
- English only, for the newsletter and all mails. No `localized` email fields.
- Subscribers stay in Resend (ADR 0005). `RESEND_AUDIENCE_ID` holds the segment id (Resend renamed Audiences to Segments; existing ids carry over).
- Newsletters have no public page (`pathOf('newsletters')` is null), so the footer has no "view in browser" link.
- The logo is an upload in the global (PNG or JPG). The site's `logo.webp` does not show in Outlook.

```ts
import type { Payload, PayloadRequest } from 'payload'

/**
 * What every email reads from the Email settings global, with a default for
 * each empty field. Texts carry {placeholders} that `fill` replaces.
 */

export const DEFAULT_TEXTS = {
  acceptance: {
    subject: 'You are accepted to {title}',
    body: 'Dear {name},\n\nYou are accepted to {title}.',
    new_account:
      'We made a CILAS account for {email}. To sign in, use “Continue with Google” if this is a Google address. If not, choose “Forgot password?” to set a password.',
    existing_account: 'Sign in with your CILAS account as usual.',
    button: 'Sign in',
  },
  review: {
    submitted_subject: 'Submitted for review: {title}',
    submitted: '“{title}” was submitted for review.',
    approved_subject: 'Published: {title}',
    approved: '“{title}” was approved and is now published.',
    changes_requested_subject: 'Changes requested: {title}',
    changes_requested: '“{title}” needs changes before it can be published.',
    rejected_subject: 'Not accepted: {title}',
    rejected: '“{title}” was not accepted for publication.',
    note_label: 'Note from the editor',
  },
  reset: {
    subject: 'Reset your CILAS password',
    body: 'Someone asked to reset the password for {email}. To choose a new password, use the button below. If this was not you, ignore this email: your password stays the same.',
    button: 'Choose a new password',
  },
  newsletter: {
    read: 'Read more',
    apply: 'Apply',
    unsubscribe: 'Unsubscribe',
  },
} as const

type Texts = { [K in keyof typeof DEFAULT_TEXTS]: { [F in keyof (typeof DEFAULT_TEXTS)[K]]: string } }

export type EmailSettings = {
  logo: { url: string; alt: string } | null
  accent: string
  fromName: string
  replyTo: string | null
  footer: string
  texts: Texts
}

/** The global as stored: any field may be empty. */
export type StoredEmailSettings = {
  logo?: unknown
  accent?: string | null
  from_name?: string | null
  reply_to?: string | null
  footer?: string | null
  texts?: { [K in keyof Texts]?: { [F in keyof Texts[K]]?: string | null } | null } | null
}

const DEFAULT_FOOTER = 'Cairo Institute for Liberal Arts and Sciences (CILAS)\nCairo, Egypt'

export const withDefaults = (stored: StoredEmailSettings): EmailSettings => {
  const texts = {} as Record<string, Record<string, string>>
  for (const [group, fields] of Object.entries(DEFAULT_TEXTS)) {
    const saved = (stored.texts as Record<string, Record<string, string | null>> | null)?.[group]
    texts[group] = {}
    for (const [field, fallback] of Object.entries(fields)) {
      texts[group][field] = saved?.[field]?.trim() || fallback
    }
  }
  const logo = stored.logo as { url?: string | null; alt?: string | null } | null | undefined
  return {
    logo: logo && typeof logo === 'object' && logo.url ? { url: logo.url, alt: logo.alt || 'CILAS' } : null,
    accent: /^#[0-9a-f]{6}$/i.test(stored.accent ?? '') ? stored.accent! : '#000000',
    fromName: stored.from_name?.trim() || 'CILAS',
    replyTo: stored.reply_to?.trim() || null,
    footer: stored.footer?.trim() || DEFAULT_FOOTER,
    texts: texts as Texts,
  }
}

/** The global, read past access, with defaults filled in. */
export const getEmailSettings = async (payload: Payload, req?: PayloadRequest) =>
  withDefaults(
    (await payload.findGlobal({
      slug: 'email-settings',
      depth: 1,
      overrideAccess: true,
      req,
    })) as StoredEmailSettings,
  )

/** Replaces each {key} with its value; unknown keys stay as written. */
export const fill = (template: string, values: Record<string, string>): string =>
  template.replace(/\{(\w+)\}/g, (match, key: string) => values[key] ?? match)

/** A textarea's text as paragraphs: blank lines separate them. */
export const paragraphs = (text: string): string[] =>
  text
    .split(/\n\s*\n/)
    .map((p) => p.trim())
    .filter(Boolean)
```

```ts
  acceptance: 'Placeholders: {name}, {email}, {title}.',
  review: 'Placeholder: {title}.',
  reset: 'Placeholder: {email}.',
  newsletter: 'Button and link labels in newsletter issues.',
}

const LABELS: Record<string, string> = {
  acceptance: 'Application accepted',
  review: 'Article review',
  reset: 'Password reset',
  newsletter: 'Newsletter',
}

/** One group per email; each field shows its default and is used when empty. */
const textGroups: Field[] = Object.entries(DEFAULT_TEXTS).map(([group, fields]) => ({
  name: group,
  label: LABELS[group],
  type: 'group',
  admin: { description: `${HINTS[group]} Leave a field empty to use the text shown in it.` },
  fields: Object.entries(fields).map(([name, fallback]): Field => {
    const long = fallback.length > 60
    return {
      name,
      type: long ? 'textarea' : 'text',
      admin: { placeholder: fallback, ...(long ? { rows: 4 } : {}) },
    }
  }),
}))
```

- [ ] **Step 5: Write `src/emails/message.tsx`**

```tsx
import { Button, Section, Text } from 'react-email'

import { EmailLayout } from './layout'
import type { EmailSettings } from './settings'
import { button as buttonStyle, colors, type } from './theme'

/** A transactional email: paragraphs, an optional quoted note, an optional button. */
export function MessageEmail({
  settings,
  preview,
  paragraphs,
  note,
  button,
}: {
  settings: EmailSettings
  preview: string
  paragraphs: string[]
  note?: { label: string; text: string } | null
  button?: { label: string; url: string } | null
}) {
  return (
    <EmailLayout settings={settings} preview={preview}>
      {paragraphs.map((p, i) => (
        <Text key={i} style={{ ...type.body, margin: '0 0 16px', whiteSpace: 'pre-line' }}>
          {p}
        </Text>
      ))}
      {note ? (
        <Section style={{ borderLeft: `3px solid ${colors.stroke}`, padding: '4px 0 4px 16px', margin: '0 0 16px' }}>
          <Text style={{ ...type.small, margin: '0 0 4px' }}>{note.label}</Text>
          <Text style={{ ...type.body, margin: 0, whiteSpace: 'pre-line' }}>{note.text}</Text>
        </Section>
      ) : null}
      {button ? (
        <Section style={{ margin: '8px 0 0' }}>
          <Button href={button.url} style={buttonStyle(settings.accent)}>
            {button.label}
          </Button>
        </Section>
      ) : null}
    </EmailLayout>
  )
}
```

```ts
import { convertLexicalToPlaintext } from '@payloadcms/richtext-lexical/plaintext'

import { pathOf } from '@/lib/paths'

/** One newsletter pick as the email shows it. */
export type Card = {
  kind: 'Course' | 'Article' | 'Programme'
  title: string
  meta: string | null
  text: string | null
  note: string | null
  image: { url: string; alt: string } | null
  url: string
  button: string
}

type Pick = { relationTo: 'courses' | 'articles' | 'programmes'; value: unknown }
type Doc = Record<string, unknown> & { id: number; title?: string; slug?: string | null }

const imageOf = (media: unknown, alt: string) => {
  const m = media as { url?: string | null; alt?: string | null } | null
  return m && typeof m === 'object' && m.url?.startsWith('http') ? { url: m.url, alt: m.alt || alt } : null
}

const SEASONS: Record<string, string> = { fall: 'Fall', winter: 'Winter', spring: 'Spring', summer: 'Summer' }

/** The start of an article's body, for its card. */
const excerpt = (body: unknown, length = 180) => {
  if (!body || typeof body !== 'object') return null
  const text = convertLexicalToPlaintext({ data: body as never }).replace(/\s+/g, ' ').trim()
  return text.length > length ? `${text.slice(0, length).replace(/\s+\S*$/, '')}…` : text || null
}

/** A populated pick as a card, or null when it is missing or has no page. */
export const cardOf = (
  pick: Pick,
  note: string | null | undefined,
  labels: { site: string; read: string; apply: string },
): Card | null => {
  const doc = pick.value as Doc | null
  if (!doc || typeof doc !== 'object' || !doc.title) return null
  const site = labels.site.replace(/\/+$/, '')
  const base = { title: doc.title, note: note?.trim() || null }

  if (pick.relationTo === 'courses') {
    const path = pathOf('courses', doc, 'en')
    if (!path) return null
    const cycle = doc.cycle as { season?: string; start_year?: number } | null
    return {
      ...base,
      kind: 'Course',
      meta: cycle && typeof cycle === 'object' && cycle.season ? `${SEASONS[cycle.season] ?? cycle.season} ${cycle.start_year ?? ''}`.trim() : null,
      text: (doc.summary as string | null) || (doc.subtitle as string | null) || null,
      image: imageOf(doc.cover_image, doc.title),
      url: `${site}${path}`,
      button: labels.read,
    }
  }
  if (pick.relationTo === 'articles') {
    const path = pathOf('articles', doc, 'en')
    if (!path) return null
    const authors = ((doc.authors as Array<{ name?: string } | number>) ?? [])
      .map((a) => (typeof a === 'object' ? a.name : null))
      .filter(Boolean)
    return {
      ...base,
      kind: 'Article',
      meta: authors.length ? authors.join(', ') : null,
      text: excerpt(doc.body),
      image: imageOf(doc.cover_image, doc.title),
      url: `${site}${path}`,
      button: labels.read,
    }
  }
  // Only the Bridge has a page of its own (/study/bridge).
  if (doc.kind !== 'bridge') return null
  return {
    ...base,
    kind: 'Programme',
    meta: null,
    text: (doc.tagline as string | null) || null,
    image: imageOf(doc.cover, doc.title),
    url: `${site}/en/study/bridge`,
    button: labels.apply,
  }
}
```

```ts
import config from '@payload-config'
import { headers as getHeaders } from 'next/headers.js'
import { getPayload } from 'payload'

import { isAdminUser } from '@/access'
import { newsletterMail } from '@/emails/newsletter'
import { getEmailSettings } from '@/emails/settings'
import { acceptanceMail, resetMail, reviewMail } from '@/emails/transactional'
import { env } from '@/env'

/**
 * Admin-only email previews for the live-preview pane: a newsletter issue
 * (its latest draft) or every transactional email with sample data. The
 * script reloads the page when Payload saves, as RefreshRouteOnSave does.
 */
export async function GET(request: Request) {
  const payload = await getPayload({ config })
  const { user } = await payload.auth({ headers: await getHeaders() })
  if (!isAdminUser(user)) return new Response('Admins only', { status: 403 })

  const url = new URL(request.url)
  const settings = await getEmailSettings(payload)
  const site = env.NEXT_PUBLIC_SERVER_URL
  let html: string

  if (url.searchParams.get('kind') === 'newsletter') {
    const id = Number(url.searchParams.get('id'))
    if (!Number.isInteger(id)) return new Response('Save the issue to see its preview.', { status: 404 })
    const issue = await payload.findByID({ collection: 'newsletters', id, depth: 2, draft: true, locale: 'en', overrideAccess: true })
    html = (await newsletterMail(issue as never, settings, { site, unsubscribe: '#unsubscribe' })).html
  } else {
    const sample = { name: 'Mona Adel', email: 'mona@example.com', site }
    const mails = await Promise.all([
      acceptanceMail(settings, { ...sample, title: 'The Bridge Programme 2026', newAccount: true }),
      reviewMail(settings, 'changes_requested', 'On translation', 'Please shorten the second part.'),
      resetMail(settings, { email: sample.email, token: 'sample-token', site }),
    ])
    html = mails
      .map((m) => `<h2 style="font:600 14px system-ui;margin:24px 16px 8px">${m.subject}</h2><iframe style="width:100%;height:560px;border:0" srcdoc="${m.html.replace(/&/g, '&amp;').replace(/"/g, '&quot;')}"></iframe>`)
      .join('')
  }

  const reload = `<script>
    window.parent.postMessage({ type: 'payload-live-preview', ready: true }, ${JSON.stringify(site)});
    window.addEventListener('message', (e) => {
      if (e.origin === ${JSON.stringify(site)} && e.data && e.data.type === 'payload-document-event') location.reload();
    });
  </script>`
  return new Response(html.replace('</body>', `${reload}</body>`).concat(html.includes('</body>') ? '' : reload), {
    headers: { 'Content-Type': 'text/html; charset=utf-8', 'Cache-Control': 'no-store' },
  })
}
```

```ts
/**
 * Sends a newsletter issue to a Resend segment (ADR 0005: the list lives in
 * Resend). Resend adds each contact's unsubscribe link where the HTML has
 * {{{RESEND_UNSUBSCRIBE_URL}}}. Returns the broadcast id.
 */
export async function sendBroadcast(args: {
  apiKey: string
  segmentId: string
  from: string
  replyTo: string | null
  subject: string
  html: string
  text: string
  name: string
}): Promise<string> {
  const res = await fetch('https://api.resend.com/broadcasts', {
    method: 'POST',
    headers: { Authorization: `Bearer ${args.apiKey}`, 'Content-Type': 'application/json' },
    body: JSON.stringify({
      segment_id: args.segmentId,
      from: args.from,
      ...(args.replyTo ? { reply_to: args.replyTo } : {}),
      subject: args.subject,
      html: args.html,
      text: args.text,
      name: args.name,
      send: true,
    }),
  })
  const body = (await res.json().catch(() => ({}))) as { id?: string; message?: string }
  if (!res.ok || !body.id) throw new Error(body.message ?? `Resend answered ${res.status}`)
  return body.id
}
```

```ts
import type { PayloadHandler, PayloadRequest } from 'payload'

import { isAdminUser } from '@/access'
import { newsletterMail } from '@/emails/newsletter'
import { getEmailSettings } from '@/emails/settings'
import { env } from '@/env'
import { sendBroadcast } from '@/lib/broadcast'

const issueOf = (req: PayloadRequest, id: number) =>
  req.payload.findByID({ collection: 'newsletters', id, depth: 2, draft: true, locale: 'en', overrideAccess: true, req })

const guarded =
  (run: (req: PayloadRequest, id: number) => Promise<Response>): PayloadHandler =>
  async (req) => {
    if (!isAdminUser(req.user)) return Response.json({ error: 'Admins only.' }, { status: 403 })
    const id = Number(req.routeParams?.id)
    if (!Number.isInteger(id)) return Response.json({ error: 'Save the issue first.' }, { status: 404 })
    try {
      return await run(req, id)
    } catch (error) {
      req.payload.logger.error({ err: error, msg: `Newsletter ${id} failed to send` })
      return Response.json({ error: (error as Error).message }, { status: 502 })
    }
  }

/** The issue as it stands, to the admin who asked. */
const test = guarded(async (req, id) => {
  const mail = await newsletterMail(await issueOf(req, id), await getEmailSettings(req.payload, req), {
    site: env.NEXT_PUBLIC_SERVER_URL,
    unsubscribe: `${env.NEXT_PUBLIC_SERVER_URL}/en`,
  })
  await req.payload.sendEmail({ to: req.user!.email, subject: `[Test] ${mail.subject}`, html: mail.html, text: mail.text })
  return Response.json({ ok: true, to: req.user!.email })
})

/** The issue to every subscriber, once; then it is published to the archive. */
const send = guarded(async (req, id) => {
  if (!env.RESEND_API_KEY || !env.RESEND_AUDIENCE_ID) {
    return Response.json({ error: 'Sending is off: RESEND_API_KEY or RESEND_AUDIENCE_ID is not set.' }, { status: 503 })
  }
  const issue = await issueOf(req, id)
  if (issue.sent_at) return Response.json({ error: 'This issue was already sent.' }, { status: 409 })
  const settings = await getEmailSettings(req.payload, req)
  const mail = await newsletterMail(issue, settings, {
    site: env.NEXT_PUBLIC_SERVER_URL,
    unsubscribe: '{{{RESEND_UNSUBSCRIBE_URL}}}',
  })
  const broadcastId = await sendBroadcast({
    apiKey: env.RESEND_API_KEY,
    segmentId: env.RESEND_AUDIENCE_ID,
    from: `${settings.fromName} <${env.EMAIL_FROM_ADDRESS}>`,
    replyTo: settings.replyTo,
    subject: mail.subject,
    html: mail.html,
    text: mail.text,
    name: issue.title,
  })
  await req.payload.update({
    collection: 'newsletters',
    id,
    data: { _status: 'published', sent_at: new Date().toISOString(), broadcast_id: broadcastId },
    overrideAccess: true,
    req,
  })
  return Response.json({ ok: true, broadcastId })
})

export const newsletterEndpoints = [
  { path: '/:id/test', method: 'post' as const, handler: test },
  { path: '/:id/send', method: 'post' as const, handler: send },
]
```
