# Boat Hero Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the home page hero with a full-screen, theme-aware ink-brush felucca scene, with CILAS in numbers along its bottom and a nav bar that stays clear while the hero is on screen.

**Architecture:** `HomeHero` (client) owns the layout, the existing copy and actions, and an IntersectionObserver that sets `data-over-hero` on `<html>`. It renders two new pieces: `HomeScene` (server-safe inline SVG + CSS module, all motion in CSS) and `HomeNumbers` (client, counts up once). One global CSS rule keeps the header (given a stable `site-header` class) transparent while `data-over-hero` is set. Scroll-linked motion uses CSS `animation-timeline: scroll(root)` behind `@supports`.

**Tech Stack:** Next.js App Router, React 19, next-intl, `motion/react`, Tailwind v4 + plain CSS in `src/app/(frontend)/styles.css`, CSS modules, vitest (unit), Playwright (e2e).

**Spec:** `docs/superpowers/specs/2026-10-17-boat-hero-design.md`

**Deviation from spec, on purpose:** the spec names `animation-timeline: view()` with `animation-range: exit`. The plan uses `scroll(root)` with `animation-range: 0 90vh` instead: the hero always sits at the top of the page, so the two are equivalent here, and `scroll(root)` does not depend on how the hero's negative top margin and the deadlines banner shift its box.

**Working rules:** Work on `main`. Commit only the files each task names: the working tree has unrelated uncommitted seed data, so never `git add -A` or `git add .`.

```tsx
'use client'

import { animate, useReducedMotion } from 'motion/react'
import { useEffect, useRef } from 'react'

import { localDigits } from '@/lib/digits'

export type HomeStat = { value: number; label: string }

/** The count starts as the row fades in under the drawn scene (see .home-numbers). */
const DELAY = 0.9
const DURATION = 1.6

/**
 * CILAS in numbers, along the bottom of the hero. The server writes the final
 * values; once hydrated, each counts up from zero while the row fades in. The
 * moving digits are hidden from screen readers, which read the final value.
 */
export function HomeNumbers({
  stats,
  title,
  locale,
}: {
  stats: HomeStat[]
  title: string
  locale: string
}) {
  const reduced = useReducedMotion()
  const values = useRef<(HTMLSpanElement | null)[]>([])
  // A primitive, so the hero's re-renders (its rotating line) don't restart the count.
  const targets = stats.map((stat) => stat.value).join(',')

  useEffect(() => {
    if (reduced) return
    const finals = targets.split(',').map(Number)
    const controls = finals.map((final, i) => {
      const el = values.current[i]
      if (!el) return null
      el.textContent = localDigits(0, locale)
      return animate(0, final, {
        delay: DELAY,
        duration: DURATION,
        ease: [0.2, 0, 0, 1],
        onUpdate: (v) => {
          el.textContent = localDigits(Math.round(v), locale)
        },
      })
    })
    return () => {
      controls.forEach((control, i) => {
        control?.stop()
        const el = values.current[i]
        if (el) el.textContent = localDigits(finals[i], locale)
      })
    }
  }, [reduced, targets, locale])

  return (
    <section aria-labelledby="home-numbers-title" className="home-numbers">
      <h2 className="sr-only" id="home-numbers-title">
        {title}
      </h2>
      <dl className="home-numbers-list">
        {stats.map((stat, i) => (
          <div className="home-number" key={stat.label}>
            <dt className="home-number-label">{stat.label}</dt>
            <dd className="home-number-value">
              <span
                aria-hidden
                ref={(el) => {
                  values.current[i] = el
                }}
              >
                {localDigits(stat.value, locale)}
              </span>
              <span className="sr-only">{localDigits(stat.value, locale)}</span>
            </dd>
          </div>
        ))}
      </dl>
    </section>
  )
}
```

```tsx
'use client'

import { AnimatePresence, motion, useReducedMotion } from 'motion/react'
import { useEffect, useRef, useState } from 'react'

import { TextMorph } from '@/components/arc/text-morph/text-morph'
import { TextReveal } from '@/components/arc/text-reveal/text-reveal'
import { HomeNumbers, type HomeStat } from '@/components/home/home-numbers'
import { HomeScene } from '@/components/home/home-scene'
import { Link } from '@/i18n/navigation'

type Action = { label: string; href: string }
/** The main action while it can't be taken: its label, and why, with where to go instead. */
type Closed = { label: string; note: string; link: Action }

/** How long each audience stays before the next takes its place. */
const STEP = 2400

/**
 * The home page's opening, the whole first screen: the motto over a felucca
 * drawn in the ink of the logo, who CILAS is for, one after another, and the
 * institute in numbers along the water. The list runs once and rests on its
 * last line; with reduced motion it starts there. While any of it is under
 * the header, the header stays clear.
 */
export function HomeHero({
  title,
  text,
  who,
  primary,
  secondary,
  numbers,
  locale,
}: {
  title: string
  text: string
  who: string[]
  /** Unavailable: the button shows but does nothing, and the note says why. */
  primary: Action | Closed
  secondary: Action
  numbers: { title: string; stats: HomeStat[] }
  locale: string
}) {
  const reduced = useReducedMotion()
  const [index, setIndex] = useState(0)
  const last = who.length - 1
  // The server renders the first line; with reduced motion the list jumps to its last line once hydrated, so both renders agree.
  const shown = index

  useEffect(() => {
    if (reduced) {
      // eslint-disable-next-line react-hooks/set-state-in-effect
      setIndex(last)
      return
    }
    if (index >= last) return
    const timer = window.setTimeout(() => setIndex((i) => i + 1), STEP)
    return () => window.clearTimeout(timer)
  }, [index, last, reduced])

  // data-over-hero on <html> while the hero shows below the header's bottom edge.
  const ref = useRef<HTMLElement>(null)
  useEffect(() => {
    const hero = ref.current
    if (!hero) return
    const root = document.documentElement
    const header = parseFloat(getComputedStyle(root).getPropertyValue('--site-header-height')) || 64
    const observer = new IntersectionObserver(
      ([entry]) => root.toggleAttribute('data-over-hero', entry.isIntersecting),
      { rootMargin: `-${header}px 0px 0px 0px` },
    )
    observer.observe(hero)
    return () => {
      observer.disconnect()
      root.removeAttribute('data-over-hero')
    }
  }, [])

  return (
    <section aria-labelledby="home-title" className="home-hero" ref={ref}>
      <HomeScene />
      <div className="home-hero-inner">
        <TextReveal as="h1" className="home-hero-title" id="home-title" text={title} />
        <p className="home-hero-text">{text}</p>
        {/* Read once, whole, by screen readers; the moving line is for the eye. */}
        <p className="sr-only">{who.join(locale === 'ar' ? '، ' : ', ')}</p>
        <p aria-hidden className="home-hero-who">
          {locale === 'ar' ? (
            // Arabic letters join, so the line changes whole instead of letter by
            // letter. Leaving and arriving lines share one grid cell and
            // crossfade in place: popLayout's absolute positioning is measured
            // from the left and threw the leaving line aside in RTL.
            <span className="home-hero-who-stack">
              <AnimatePresence initial={false}>
                <motion.span
                  animate={{ opacity: 1, filter: 'blur(0px)' }}
                  exit={{ opacity: 0, filter: 'blur(4px)' }}
                  initial={{ opacity: 0, filter: 'blur(4px)' }}
                  key={shown}
                  transition={{ duration: 0.35, ease: [0.2, 0, 0, 1] }}
                >
                  {who[shown]}
                </motion.span>
              </AnimatePresence>
            </span>
          ) : (
            <TextMorph>{who[shown] ?? ''}</TextMorph>
          )}
        </p>
        <div className="home-hero-actions">
          {'href' in primary ? (
            <Link className="home-action" data-primary href={primary.href}>
              {primary.label}
            </Link>
          ) : (
            // aria-disabled rather than disabled: it stays reachable, so keyboard and screen-reader users meet the note too.
            <button
              aria-describedby="home-hero-note"
              aria-disabled
              className="home-action"
              data-primary
              type="button"
            >
              {primary.label}
            </button>
          )}
          <Link className="home-action" href={secondary.href}>
            {secondary.label}
          </Link>
        </div>
        {'note' in primary ? (
          <p className="home-hero-note" id="home-hero-note">
            {primary.note} <a href={primary.link.href}>{primary.link.label}</a>
          </p>
        ) : null}
      </div>
      <HomeNumbers locale={locale} stats={numbers.stats} title={numbers.title} />
    </section>
  )
}
```

import { expect, test } from '@playwright/test'

const BASE = 'http://localhost:3000'

const scrollTo = (top: number) => window.scrollTo({ top, behavior: 'instant' })

for (const locale of ['en', 'ar']) {

    await page.evaluate(scrollTo, await page.evaluate(() => window.innerHeight * 1.5))
    await expect(header).not.toHaveCSS('background-color', CLEAR)

    await page.evaluate(scrollTo, 0)
    await expect(header).toHaveCSS('background-color', CLEAR)
  })
}

test('elsewhere the header turns solid after a small scroll, as before', async ({ page }) => {
  await page.goto(`${BASE}/en/articles`)
  const header = page.locator('.site-header')
  await page.evaluate(scrollTo, 400)
  await expect(header).not.toHaveCSS('background-color', CLEAR)
})
